/** Muncul halus: blur -> tajam, naik sedikit. */
const useRise = (start: number, dur = 20) => {
  const frame = useFrame();
  const { fps } = useVideoConfig();
  return spring({ frame: frame - start, fps, config: { damping: 200 }, durationInFrames: dur });
};
const rise = (p: number, dy = 40): React.CSSProperties => ({
  opacity: p,
  filter: `blur(${(1 - p) * 14}px)`,
  transform: `translateY(${(1 - p) * dy}px)`,
});

const Check: React.FC<{ size?: number; color?: string; bg?: string; s?: number }> = ({ size = 44, color = C.accent, bg = C.accentSoft, s = 1 }) => (

/** Ketukan hook, diselaraskan dengan jeda di public/vo/01-hook.mp3. */
export const HOOK = { p2: 103, p3: 189 }; // diisi Promo.tsx dari timestamp kata "padahal" & "Kok"
const POST_TIMES = ["07.00", "12.00", "19.00"];

export const Hook: React.FC<SceneProps> = () => {
  const frame = useFrame();
  const { fps } = useVideoConfig();
  const id = useFilterId();
  const phase = frame < HOOK.p2 ? 1 : frame < HOOK.p3 ? 2 : 3;
  const start = phase === 1 ? 0 : phase === 2 ? HOOK.p2 : HOOK.p3;
  const end = phase === 1 ? HOOK.p2 : phase === 2 ? HOOK.p3 : 9999;
  // tiap ketukan: sapu masuk dari kanan, sapu keluar ke kiri (motion blur)
  const pin = interpolate(frame, [start, start + 8], [0, 1], { ...clamp, easing: easeOut });
  const pout = interpolate(frame, [end - 6, end], [0, 1], { ...clamp, easing: easeInOut });
  const slam = interpolate(frame, [HOOK.p3, HOOK.p3 + 6], [0, 1], { ...clamp, easing: easeOut });
  const shake = phase === 3 && frame > HOOK.p3 + 5 && frame < HOOK.p3 + 16 ? Math.sin(frame * 2.7) * (HOOK.p3 + 16 - frame) * 1.8 : 0;
  const bx = phase === 3 ? (1 - slam) * 40 : (1 - pin) * 60 + pout * 60;
  const tx = phase === 3 ? shake : (1 - pin) * 700 - pout * 700;
  return (
    <AbsoluteFill style={{ ...center }}>
      <BlurFilter id={id} x={bx} y={phase === 3 ? (1 - slam) * 30 : 0} />
      <div style={{ ...center, flexDirection: "column", filter: bx > 0.3 ? `url(#${id})` : undefined, transform: `translate(${tx}px, ${shake * 0.6}px)` }}>
        {phase === 1 ? (
          <>
            <Words size={104} start={2} stagger={3} segs={[{ t: "Akun Threads ini" }]} />
            <Words size={168} start={14} stagger={4} style={{ marginTop: 4 }} segs={[{ t: "posting setiap hari.", grad: GRADIENT }]} />
            <div style={{ display: "flex", gap: 20, marginTop: 44 }}>
              {POST_TIMES.map((t, i) => {
                const p = spring({ frame: frame - 40 - i * 12, fps, config: { damping: 12, stiffness: 170 } });
                return (
                  <div
                    key={t}
                    style={{
                      transform: `scale(${p}) translateY(${(1 - p) * 30}px)`,
                      opacity: Math.min(1, p * 1.5),
                      display: "flex",
                      alignItems: "center",
                      gap: 12,
                      padding: "14px 28px",
                      borderRadius: 999,
                      background: "#fff",
                      boxShadow: "0 16px 40px rgba(20,23,43,0.14)",
                      fontFamily: SANS,
                      fontWeight: 700,
                      fontSize: 32,
                      color: C.ink,
                    }}
                  >
                    <span style={{ width: 14, height: 14, borderRadius: "50%", background: "#0f9d6c", boxShadow: "0 0 12px #3dd6a0" }} />
                    Terbit {t}
                  </div>
                );
              })}
            </div>
          </>
        ) : phase === 2 ? (
          <>
            <Words size={104} start={HOOK.p2 + 4} stagger={3} segs={[{ t: "padahal yang punya" }]} />
            <div style={{ display: "flex", alignItems: "center", gap: 30, marginTop: 4 }}>
              <Words size={176} start={HOOK.p2 + 20} stagger={4} segs={[{ t: "lagi tidur.", grad: GRADIENT }]} />
              <div style={{ position: "relative", width: 150, height: 150, opacity: interpolate(frame, [HOOK.p2 + 26, HOOK.p2 + 34], [0, 1], clamp) }}>
                <svg width="130" height="130" viewBox="0 0 24 24" style={{ filter: `drop-shadow(0 0 24px ${C.accentLight})` }}>
                  <path d="M20 14.5A8.5 8.5 0 0 1 9.5 4a8.5 8.5 0 1 0 10.5 10.5z" fill={C.accent} />
                </svg>
                {[0, 1, 2].map((i) => {
                  const t = (frame - HOOK.p2 - 30 - i * 9) / 30;
                  const k = t - Math.floor(t);
                  return (
                    <div
                      key={i}
                      style={{
                        position: "absolute",
                        left: 110 + k * 50 + i * 8,
                        top: 10 - k * 90,
                        fontFamily: DISPLAY,
                        fontWeight: 800,
                        fontSize: 34 + i * 10,
                        color: C.accent,
                        opacity: t < 0 ? 0 : Math.sin(k * Math.PI),
                      }}
                    >
                      z
                    </div>
                  );
                })}
              </div>
            </div>
          </>
        ) : (
          <div
            style={{
              transform: `scale(${2.6 - 1.6 * slam})`,
              opacity: Math.min(1, slam * 1.5),
              fontFamily: DISPLAY,
              fontWeight: 800,
              fontSize: 260,
              letterSpacing: "-0.05em",
              color: C.ink,
              whiteSpace: "nowrap",
            }}
          >
            Kok <GradText>bisa?</GradText>
          </div>
        )}
      </div>
    </AbsoluteFill>
  );
};

export const BigText: React.FC<SceneProps> = () => {
  const frame = useFrame();
  const id = useFilterId();
  const keys = [0, 22, 40, 58, 80];
  const camAt = (f: number) => ({
    s: interpolate(f, keys, [1.6, 1.6, 4.4, 4.4, 1.1], { ...clamp, easing: easeInOut }),
    x: interpolate(f, keys, [430, 430, -20, -560, 0], { ...clamp, easing: easeInOut }),
  });
  const now = camAt(frame);
  const prev = camAt(frame - 1);
  const v = Math.abs(now.x * now.s - prev.x * prev.s);
  const bx = Math.min(55, v * 0.45);
  return (
    <AbsoluteFill style={{ ...center }}>
      <BlurFilter id={id} x={bx} y={0} />
      <div style={{ transform: `scale(${now.s}) translateX(${now.x}px)`, filter: bx > 0.3 ? `url(#${id})` : undefined }}>
        <div style={{ display: "flex", columnGap: 34 }}>
          <Words size={124} start={2} stagger={4} segs={[{ t: "Dari ide" }]} />
          <Words size={124} start={26} stagger={5} segs={[{ t: "sampai terbit.", grad: GRADIENT }]} />
        </div>
      </div>
    </AbsoluteFill>
  );
};
